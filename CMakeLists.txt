cmake_minimum_required(VERSION 3.28)
project(cam_imu_calib VERSION 0.1.0 LANGUAGES C CXX)
set(CALIB_CXX_STANDARD 26 CACHE STRING "C++ standard (26, or explicit fallback 23)")
if(NOT CALIB_CXX_STANDARD MATCHES "^(23|26)$")
  message(FATAL_ERROR "CALIB_CXX_STANDARD must be 23 or 26")
endif()
set(CMAKE_CXX_STANDARD ${CALIB_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CALIB_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(CALIB_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(PNG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
find_package(CLI11 2.3 CONFIG REQUIRED)
find_package(yaml-cpp 0.8 CONFIG REQUIRED)
find_package(nlohmann_json 3.11 CONFIG REQUIRED)
include(cmake/AprilTag.cmake)
include(cmake/OpenCV.cmake)
include(cmake/Ceres.cmake)
add_library(calib_camera_models src/camera_model.cpp)
target_include_directories(calib_camera_models PUBLIC include)
target_compile_options(calib_camera_models PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_library(calib_core src/dataset.cpp src/config.cpp src/detection.cpp src/observations.cpp)
target_include_directories(calib_core PUBLIC include)
target_link_libraries(calib_core PUBLIC calib_camera_models nlohmann_json::nlohmann_json PRIVATE PNG::PNG ZLIB::ZLIB yaml-cpp::yaml-cpp apriltag Threads::Threads)
target_compile_options(calib_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_compile_definitions(calib_core PUBLIC CALIB_VERSION="${PROJECT_VERSION}" CALIB_APRILTAG_VERSION="${CALIB_APRILTAG_VERSION}" CALIB_APRILTAG_COMMIT="${CALIB_APRILTAG_COMMIT}")
add_library(calib_initialization src/initialization.cpp src/initialization_io.cpp)
target_link_libraries(calib_initialization PUBLIC calib_core PRIVATE opencv_calib opencv_geometry yaml-cpp::yaml-cpp)
target_include_directories(calib_initialization SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(calib_initialization PRIVATE CALIB_OPENCV_VERSION="${CALIB_OPENCV_VERSION}")
target_compile_options(calib_initialization PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_library(calib_rig src/rig.cpp src/rig_io.cpp)
target_link_libraries(calib_rig PUBLIC calib_initialization PRIVATE Ceres::ceres yaml-cpp::yaml-cpp)
target_compile_options(calib_rig PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
foreach(command IN ITEMS calib_cameras calib_imu_camera)
  add_executable(${command} src/main.cpp)
  target_link_libraries(${command} PRIVATE calib_rig CLI11::CLI11)
  target_compile_options(${command} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
target_compile_definitions(calib_imu_camera PRIVATE CALIB_REQUIRE_IMU=1)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
  add_executable(calib_tests tests/unit.cpp)
  target_link_libraries(calib_tests PRIVATE calib_core apriltag PNG::PNG)
  add_test(NAME core COMMAND calib_tests)
  add_executable(observation_tests tests/observations.cpp)
  target_link_libraries(observation_tests PRIVATE calib_core)
  add_test(NAME observations COMMAND observation_tests)
  add_executable(camera_model_tests tests/camera_model.cpp)
  target_link_libraries(camera_model_tests PRIVATE calib_core)
  add_test(NAME camera_models COMMAND camera_model_tests)
  add_executable(initialize_test_camera tests/initialize_camera.cpp)
  target_link_libraries(initialize_test_camera PRIVATE calib_initialization)
  add_test(NAME initialization COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/initialization.py $<TARGET_FILE:initialize_test_camera>)
  set_tests_properties(initialization PROPERTIES TIMEOUT 240)
  add_executable(initialization_unit_tests tests/initialization_unit.cpp)
  target_link_libraries(initialization_unit_tests PRIVATE calib_initialization)
  add_test(NAME initialization_selection COMMAND initialization_unit_tests)
  add_executable(rig_residual_tests tests/rig_residual.cpp)
  target_link_libraries(rig_residual_tests PRIVATE calib_rig Ceres::ceres)
  add_test(NAME rig_derivatives COMMAND rig_residual_tests)
  add_executable(rig_test_bridge tests/rig_bridge.cpp)
  target_link_libraries(rig_test_bridge PRIVATE calib_rig)
  find_program(PDFINFO_EXECUTABLE pdfinfo REQUIRED)
  find_program(PDFTOTEXT_EXECUTABLE pdftotext REQUIRED)
  add_test(NAME rig_calibration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/rig.py $<TARGET_FILE:rig_test_bridge> ${PDFINFO_EXECUTABLE} ${PDFTOTEXT_EXECUTABLE})
  set_tests_properties(rig_calibration PROPERTIES TIMEOUT 300)
  add_executable(project_test_camera tests/project_camera.cpp)
  target_link_libraries(project_test_camera PRIVATE calib_core)
  option(CALIB_OPENCV_TESTS "Compare optical models against Python OpenCV (test dependency only)" OFF)
  if(CALIB_OPENCV_TESTS)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import cv2, numpy"
      RESULT_VARIABLE opencv_import_status ERROR_QUIET)
    if(NOT opencv_import_status EQUAL 0)
      message(FATAL_ERROR "CALIB_OPENCV_TESTS requires cv2 and numpy for ${Python3_EXECUTABLE}; see INSTALLATION.md")
    endif()
    add_test(NAME camera_models_opencv COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/camera_model_oracle.py $<TARGET_FILE:project_test_camera>)
  endif()
  add_executable(render_test_tags tests/render_tags.cpp)
  target_link_libraries(render_test_tags PRIVATE apriltag PNG::PNG)
  add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:calib_cameras> $<TARGET_FILE:calib_imu_camera> $<TARGET_FILE:render_test_tags>)
  add_executable(decode_test_image tests/decode_image.cpp)
  target_link_libraries(decode_test_image PRIVATE calib_core)
  add_test(NAME png16 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/png16.py $<TARGET_FILE:decode_test_image>)
  add_test(NAME dependency_lock COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/dependencies.py)
  set_tests_properties(cli_integration PROPERTIES TIMEOUT 120)
  if(CALIB_SANITIZERS)
    set_tests_properties(core observations camera_models initialization initialization_selection rig_derivatives rig_calibration cli_integration png16 PROPERTIES ENVIRONMENT
      "UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1;ASAN_OPTIONS=detect_leaks=1:halt_on_error=1")
    if(CALIB_OPENCV_TESTS)
      set_tests_properties(camera_models_opencv PROPERTIES ENVIRONMENT
        "UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1;ASAN_OPTIONS=detect_leaks=1:halt_on_error=1")
    endif()
  endif()
endif()
